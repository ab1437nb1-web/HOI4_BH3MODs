### 第一个律者在崩坏出现后2天创建 ###
country_event = {
	id = HerrscherCome.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		set_temp_variable = {  temp_BH3_herrscher_time =  global.BH3_herrscher_time }
		set_variable = { global.BHS_Herrscher_Landing_times = temp_BH3_herrscher_time }
		BHS_PreLanding = yes
		clear_variable = temp_BH3_herrscher_time
	}
}
### 理之律者 ###
#创建国家
country_event = {
	id = HerrscherCome.100
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		set_variable = { global.BHS_HI1_Power_Effect1 = -0.5 }#扣科研
		set_variable = { global.BHS_HI1_Power_Effect2 = -0.1 }#扣生产
		var:global.BHSLandingPoint = {
			add_core_of = HI1
			transfer_state_to = HI1
			add_dynamic_modifier = {
				modifier = BHS_High_Honkai_area 
				scope = HI1
			}
		}
		HI1 = { 
			set_major = yes
			set_capital = { state = var:global.BHSLandingPoint }
			#重设领导人头像和名字
			var:country_leader = {
				meta_effect = {
					text = {
						set_character_name = HI1_country_leader
						set_portraits = { civilian = { large = GFX_BH3_unit_leader_SRHerrscherofReason } }
					}
					debug = yes
				}
			}
			load_oob = "BHS_UnitPool"
			#禁止自由出兵
			country_lock_all_division_template = yes
			add_threat = 5
			#天使轮投资
			add_manpower = 6000000
			add_fuel = 999999
			# 避免部分脑瘫mod替换了country文件夹
			set_technology = {
				SishiBHS_Research = 1
				Tujin1BHS_Research = 1
				Qishi1BHS_Research = 1
				Zhanche1BHS_Research = 1
				Shengdian1BHS_Research = 1
				JianisaBHS_Research = 1
				BeinalesiBHS_Research = 1
			}
			add_equipment_to_stockpile = { type = TujinBHS_equipment_0 amount = 50000 }
			add_equipment_to_stockpile = { type = QishiBHS_equipment_0 amount = 50000 }
			add_equipment_to_stockpile = { type = ZhancheBHS_equipment_0 amount = 50000 }
			add_equipment_to_stockpile = { type = ShengdianBHS_equipment_0 amount = 50000 }
			#加入阵营
			BHS = {
				add_to_faction = HI1
			}
			every_country = {
				limit = { has_war_with = BHS }
				HI1 = { 
					declare_war_on = { target = PREV type = annex_everything } 
				}
			}
			### 崩坏阶段 ###
			#阶段1
			if = {
				limit = { BHS = { has_idea = BHS_level1_idea } } 
				add_ideas = BHS_level1_idea
			}
			#阶段2
			if = {
				limit = { BHS = { has_idea = BHS_level2_idea } } 
				add_ideas = BHS_level2_idea
			}
			#阶段3
			if = {
				limit = { BHS = { has_idea = BHS_level3_idea } } 
				add_ideas = BHS_level3_idea
			}
			#阶段4
			if = {
				limit = { BHS = { has_idea = BHS_level4_idea } } 
				add_ideas = BHS_level4_idea
			}
			#阶段5
			if = {
				limit = { BHS = { has_idea = BHS_level5_idea } } 
				add_ideas = BHS_level5_idea
			}

			### 崩坏活性 ###
			#阶段1
			if = {
				limit = { BHS = { has_idea = BHS_aggressive1 } } 
				add_ideas = BHS_aggressive1
			}
			#阶段2
			if = {
				limit = { BHS = { has_idea = BHS_aggressive2 } } 
				add_ideas = BHS_aggressive2
			}
			#阶段3
			if = {
				limit = { BHS = { has_idea = BHS_aggressive3 } } 
				add_ideas = BHS_aggressive3
			}

			### 崩坏律者单位加强 ###
			#加强
			if = {
				limit = { check_variable = {global.BH3_invasion_menu_herrscher_enhance1_btn_frame = 2 } }
				add_ideas = BHS_herrscher_enhance_idea
			}

			### 核熔蚀抗性 ###
			if = {
				limit = { BHS = { has_dynamic_modifier = { modifier = BHS_NuclearCorrosionResistance_effect } } } 
				add_dynamic_modifier = { modifier = BHS_NuclearCorrosionResistance_effect }
			}
	
			### 杂项，多选 ###
			#崩坏加强
			if = {
				limit = { BHS = { has_idea = BHS_aggressive0 } } 
				add_ideas = BHS_aggressive0
			}
			#理律权能
			add_ideas = BHS_HI1_Power
			BHS = { add_ideas = BHS_HI1_Power }
			#理律权能对人类影响
			every_country = {
				limit = { NOT = { is_in_faction_with = BHS } }
				add_dynamic_modifier = {
					modifier = BHS_HI1_Power_EffectHuman
				}
			}
			#给将领
			BH3_create_HI1_leader = yes
			#情报人员
			BH3_create_BHS_operative_leader = yes
			BHS = { BH3_create_HI1_leader = yes }
			#刷开局兵
			BHS_StartTroop_Herrscher = yes
			#给律者师
			set_technology = {
				Bronya_HerrscherofReason_Research = 1
			}
			division_template = {
				name = "the Herrscher of Reason"
				is_locked = no
				regiments = {
					HerrscherReasonBronya = { x = 0 y = 0 }
				}
			}
			var:global.BHSLandingPoint = {
				create_unit = {
					division = "name = \"the Herrscher of Reason\" division_template = \"the Herrscher of Reason\"  start_experience_factor = 1 "
					allow_spawning_on_enemy_provs = yes
					owner = HI1
					count = 1
				}	
			}
		}
		#打开超事件
		every_country = {
			limit = { is_ai = no }
			set_country_flag = BHS_Event_HI1_Come
			random_list = {
				33 = { scoped_sound_effect  = HI_Come1 }
				33 = { scoped_sound_effect  = HI_Come2 }
				33 = { scoped_sound_effect  = HI_Come3 }
			}
			country_event = { id = HerrscherCome.101 days = 3 }
		}
	}
}
