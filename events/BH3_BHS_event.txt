add_namespace = BHS
#创建崩坏国家的事件
country_event = {
	id = BHS.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		set_global_flag = BHS_Started
		set_variable = { global.HonkaiImpact_Beat_Country_Num = 0 } #统计崩坏阵营打败国家数量
		set_variable = { global.HonkaiImpact_Casualty = 0 } #统计崩坏自身伤害人数（万人）
		set_variable = { global.HonkaiImpact_Cause_Casualty = 0 } #统计崩坏造成军队伤亡（万人）
		set_variable = { global.HonkaiImpact_Population_Total = 0 } #统计人口（百万人）
		set_variable = { global.HonkaiImpact_State_Total = 0 } #统计全球地块
		every_state = {
			set_temp_variable = {BH3_state_population_m = THIS.state_population_k }
			divide_temp_variable = { BH3_state_population_m = 1000 } 
			add_to_variable = { global.HonkaiImpact_Population_Total = BH3_state_population_m } 
		} 
		round_variable = global.HonkaiImpact_Population_Total
		BH3_BHS_start = yes
		every_country = {
			limit = { is_ai = no }
			#触发爆发事件
			set_country_flag = BHS_Start_event
			play_song = "Befall"
			country_event = { id = BHS.10 days = 1 }
		}
		var:global.BHS_StartLocation = {
			set_variable = { global.HonkaiImpact_Cause_Population_Casualty = state_population_k } #统计崩坏造成人类伤亡（百万人）
		}
		divide_variable = { global.HonkaiImpact_Cause_Population_Casualty = 1000 } 
		set_variable = { global.HonkaiImpact_Remain_Population_Casualty = global.HonkaiImpact_Population_Total } #统计剩余人口（百万人）
		subtract_from_variable = { global.HonkaiImpact_Remain_Population_Casualty = HonkaiImpact_Cause_Population_Casualty }
		# 计算侵蚀人口百分比
		set_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = global.HonkaiImpact_Cause_Population_Casualty }
		divide_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = global.HonkaiImpact_Population_Total }
		multiply_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = 100 }
	}
}
