# 往事乐土：刻印动态修正
# 设计：每个英桀（系）只用一个动态修正，数值由变量累加，避免 100+ 国家精神
# 变量命名：BH3_ER_<系>_atk / _bth / _def / _org

BH3_ER_dm_dlv = {
	enable = {
		OR = {
			check_variable = { BH3_ER_dlv_window > 0 }
			has_country_flag = BH3_ER_dlv_permanent
		}
	}

	army_attack_factor = BH3_ER_dlv_atk
	army_defence_factor = BH3_ER_dlv_def
	breakthrough_factor = BH3_ER_dlv_bth
	army_org_regain = BH3_ER_dlv_org
	army_attack_against_major_factor = BH3_ER_dlv_maj
	army_attack_against_minor_factor = BH3_ER_dlv_min
}
