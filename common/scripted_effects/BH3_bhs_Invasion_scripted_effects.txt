#动态buff更新，每周更新
BH3_DynamicBuff_upgrade = {
	#崩坏攻防随浓度更新
	set_temp_variable = { temp_BHS_Atteack = global.threat }
	divide_temp_variable = { temp_BHS_Atteack = 2 }
	set_variable = { global.BHS_Atteack = temp_BHS_Atteack }
	set_temp_variable = { temp_BHS_Defend = global.threat }
	divide_temp_variable = { temp_BHS_Defend = 2 }
	set_variable = { global.BHS_Defend = temp_BHS_Defend }
	set_temp_variable = { temp_BHS_army_morale_factor = global.threat }
	multiply_temp_variable = { temp_BHS_army_morale_factor = 10 }
	set_variable = { global.BHS_army_morale_factor = temp_BHS_army_morale_factor }

	#崩坏随时间增强
	add_to_variable = { global.BHS_TacticalAtteack = 0.005 }
	add_to_variable = { global.BHS_TacticalDefend = 0.01 }

	#崩坏随文明增强
	set_temp_variable = { temp_BHS_major_country_num = 0 }
	set_temp_variable = { temp_BHS_major_country_atk = 0 }
	set_temp_variable = { temp_BHS_major_country_def = 0 }
	set_temp_variable = { temp_BHS_major_country_hp = 0 }
	set_temp_variable = { temp_BHS_major_country_org = 0 }
	every_country = {
		limit = { 
			OR = {
				is_ai = no
				is_major = yes
				NOT = { is_in_faction_with = BHS }
			}
		}
		add_to_temp_variable = { temp_BHS_major_country_num = 1 }
		add_to_temp_variable = { temp_BHS_major_country_atk = modifier@army_attack_factor }
		add_to_temp_variable = { temp_BHS_major_country_def = modifier@army_defence_factor }
		add_to_temp_variable = { temp_BHS_major_country_hp = modifier@army_strength_factor }
		add_to_temp_variable = { temp_BHS_major_country_org = modifier@army_org_factor }
	}
	divide_temp_variable = { temp_BHS_major_country_atk = temp_BHS_major_country_num }
	divide_temp_variable = { temp_BHS_major_country_def = temp_BHS_major_country_num }
	divide_temp_variable = { temp_BHS_major_country_hp = temp_BHS_major_country_num }
	divide_temp_variable = { temp_BHS_major_country_org = temp_BHS_major_country_num }
	if = {
		limit = { check_variable = { temp_BHS_major_country_atk < 0.5 } }
		set_variable = { global.BHS_CivilizationImprovement_ATK = 0.5 }
	}
	else = { set_variable = { global.BHS_CivilizationImprovement_ATK = temp_BHS_major_country_atk } }
	if = {
		limit = { check_variable = { temp_BHS_major_country_def < 0.5 } }
		set_variable = { global.BHS_CivilizationImprovement_DEF = 0.5 }
	}
	else = { set_variable = { global.BHS_CivilizationImprovement_DEF = temp_BHS_major_country_def } }
	if = {
		limit = { check_variable = { temp_BHS_major_country_hp < 0.25 } }
		set_variable = { global.BHS_CivilizationImprovement_HP = 0.25 }
	}
	else = { set_variable = { global.BHS_CivilizationImprovement_HP = temp_BHS_major_country_hp } }
	if = {
		limit = { check_variable = { temp_BHS_major_country_org < 0.25 } }
		set_variable = { global.BHS_CivilizationImprovement_ORG = 0.25 }
	}
	else = { set_variable = { global.BHS_CivilizationImprovement_ORG = temp_BHS_major_country_org } }	

	#崩坏按全球工业计算，超2000部分按0.001%计算
	set_temp_variable = { temp_IC = 0 }
	every_country = {
		add_to_temp_variable = { temp_IC = this.num_of_factories }
	}
	if = {
		limit = { check_variable = { temp_IC > 2000 } }
		set_temp_variable = { temp_IC1 = temp_IC }
		subtract_from_temp_variable = { temp_IC1 = 2000 }
		divide_temp_variable = { temp_IC1 = 1000 }
		set_temp_variable = { temp_IC = temp_IC1 }
	}
	else = {
		set_temp_variable = { temp_IC = 0 }
	}
	set_variable = { global.BHS_IC = temp_IC }

	#女武神战力加成
	set_temp_variable = { temp_BH3_Valkyrie_Score1 = global.BH3_World_Valkyire_Score }
	divide_temp_variable = { temp_BH3_Valkyrie_Score1 = 100 }
	set_variable = { global.BH3_Valkyrie_Score_effect1 = temp_BH3_Valkyrie_Score1 }
	set_temp_variable = { temp_BH3_Valkyrie_Score2 = global.BH3_World_Valkyire_Score }
	divide_temp_variable = { temp_BH3_Valkyrie_Score2 = 500 }
	set_variable = { global.BH3_Valkyrie_Score_effect2 = temp_BH3_Valkyrie_Score2 }

	#崩坏浓度对人类的影响
	#消费期望值

	divide_temp_variable = { temp_HUM_ConsumerExpected = 10 }

	divide_temp_variable = { temp_HUM_PoliticalPpower = -5 }
	set_variable = { global.HUM_PoliticalPpower = temp_HUM_PoliticalPpower }
	#动员速度
	set_temp_variable = { temp_HUM_MobilizationSpeed = global.threat }
	divide_temp_variable = { temp_HUM_MobilizationSpeed = -5 }
	set_variable = { global.HUM_MobilizationSpeed = temp_HUM_MobilizationSpeed }

#建立崩坏国家
BH3_BHS_start = {
	#取消军事禁区
	every_state = {
		set_demilitarized_zone = no
	}
	#取消原版美洲门罗主义
	every_country = {
		limit= { has_idea = USA_monroe_doctrine_idea }
		remove_ideas = USA_monroe_doctrine_idea
	}
	### 崩坏爆发地点 ###
	#随机地点
	if = {
		limit = { check_variable = {global.BH3_invasion_menu_postion1_btn_frame = 2 } }
		random_state = {
			limit = {
				state_population_k > 1000
				# industrial_complex > 1
				controller = { NOT = { TAG = SPR } }
				is_island_state = no
				# is_coastal = no
				# 
				# NOT = {
				# 	any_neighbor_state = {
				# 		is_coastal = yes
				# 	}
				# }
			}
			set_variable = { global.BHS_StartLocation = THIS }
			add_building_construction = {
				type = air_base
				level = 10
				instant_build = yes
			}
			every_country = {
				limit = { is_ai = no}
				goto_state = PREV
			}
		}
	}
	#柏林
	if = {
		limit = { check_variable = {global.BH3_invasion_menu_postion2_btn_frame = 2 } }
		set_variable = { global.BHS_StartLocation = 64 }
		64 = {
			add_building_construction = {
				type = air_base
				level = 10
				instant_build = yes
			}
		}
		every_country = {
			limit = { is_ai = no}
			goto_state = 64
		}
	}
	#西伯利亚
	if = {
		limit = { check_variable = {global.BH3_invasion_menu_postion3_btn_frame = 2 } }
		set_variable = { global.BHS_StartLocation = 576 }
		576 = {
			add_building_construction = {
				type = air_base
				level = 10
				instant_build = yes
			}
		}
		every_country = {
			limit = { is_ai = no}
			goto_state = 576
		}
	}
	#东京
	if = {
		limit = { check_variable = {global.BH3_invasion_menu_postion4_btn_frame = 2 } }
		set_variable = { global.BHS_StartLocation = 282 }
		282 = {
			add_building_construction = {
				type = air_base
				level = 10
				instant_build = yes
			}
		}
		every_country = {
			limit = { is_ai = no}
			goto_state = 282
		}
	}
	#释放国家
	var:global.BHS_StartLocation = {
		add_core_of = BHS
		transfer_state_to = BHS
		add_dynamic_modifier = { modifier = BHSinvasion_effect }
		add_dynamic_modifier = {
			modifier = BHS_High_Honkai_area 
			scope = BHS
		}
	}
	#崩坏开始
	BHS = { 
		set_capital = { state = var:global.BHS_StartLocation }
		#重设领导人头像和名字
		var:country_leader = {
			meta_effect = {
				text = {
					set_character_name = BHS_country_leader
					set_portraits = { civilian = { large = GFX_Portrait_BHS } }
				}
				log = ""
				# debug = yes #在game.log中查看meta效果
			}
		}
		add_threat = 5 
		load_oob = "BHS_UnitPool"
		#天使轮投资
		add_manpower = 6000000
		add_fuel = 999999
		# 避免部分脑瘫mod替换了country文件夹
		set_technology = {
			SishiBHS_Research = 1
			Tujin1BHS_Research = 1
			Qishi1BHS_Research = 1
			Zhanche1BHS_Research = 1
			Shengdian1BHS_Research = 1
			JianisaBHS_Research = 1
			BeinalesiBHS_Research = 1
		}
		add_equipment_to_stockpile = { type = TujinBHS_equipment_0 amount = 50000 }
		add_equipment_to_stockpile = { type = QishiBHS_equipment_0 amount = 50000 }
		add_equipment_to_stockpile = { type = ZhancheBHS_equipment_0 amount = 50000 }
		add_equipment_to_stockpile = { type = ShengdianBHS_equipment_0 amount = 50000 }

			BHS = { 
				declare_war_on = { target = PREV type = annex_everything } 
			}
		}
		every_state = {
			add_claim_by = BHS
		}

#登陆警告
BHS_PreLanding = {
	clear_variable = global.BHSLandingPoint
	clear_variable = global.BHSLandingCountry
	# 随机登陆天数
	randomize_variable = {
		var = global.BHS_Landing_Left_Day
		distribution = uniform
		min = 1
		max = 25
	}
	round_variable = global.BHS_Landing_Left_Day
	### 如果登陆次数达到律者降临计数，本次触发律者登陆 ###
	if = {
		limit = { 
			check_variable = { global.BHS_Herrscher_Landing_times = global.BH3_herrscher_time } 
			check_variable = { global.BHS_Herrscher_Beaten_num < 5 } 
		}
		random_state = {
			limit = {
				state_population_k > 1000
				controller = {
					NOT = { TAG = BHS }
					NOT = { TAG = HI1 }
					NOT = { TAG = HI2 }
					NOT = { TAG = HI3 }
					NOT = { TAG = HI6 }
					NOT = { TAG = HI7 }
					NOT = { TAG = HH2 }
				}
			}
			controller = { 
				set_variable = { global.BHSLandingPoint = PREV }
			}
			#触发律者登陆
			set_variable = { global.BHSLandingCountry = controller }
			state_event = { id = BHSstate.2 trigger_for = controller days = global.BHS_Landing_Left_Day }
			#登陆间隔计数清零
			set_variable = { global.BHS_Herrscher_Landing_times = 0 }
			#律者计数加1
			add_to_variable = { global.BHS_Herrscher_Beaten_num = 1 } 
			every_country = {
				limit = { is_ai = no}
				goto_state = PREV
			}
		}
	}
	else =  {
		random_state = {
			limit = {
				controller = { has_war_with = BHS }
				OR = {
					is_coastal = yes
					is_island_state = yes
				}
			}
			controller = { 
				set_variable = { global.BHSLandingPoint = PREV }
			}
			#触发登陆
			set_variable = { global.BHSLandingCountry = controller }
			state_event = { id = BHSstate.1 trigger_for = controller days = global.BHS_Landing_Left_Day }
			every_country = {
				limit = { is_ai = no}
				goto_state = PREV
			}
		}
	}
	every_country = {
		#所有国家触发登陆预警
		country_event = { id = BHS.11 }
		#触发X天后的登陆警告
		country_event = { id = BHS.12 days = global.BHS_Landing_Left_Day }
	}
}
