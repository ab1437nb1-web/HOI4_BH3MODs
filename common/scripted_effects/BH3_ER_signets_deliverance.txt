# 往事乐土·救世：必杀技窗口与核心II永久化
# 窗口剩余天数存在 BH3_ER_dlv_window，由 on_daily 递减；核心II解锁后置 BH3_ER_dlv_permanent
BH3_ER_dlv_ult = {
	add_to_variable = { BH3_ER_dlv_ult_count = 1 }
	if = {
		limit = { has_country_flag = BH3_ER_own_dlv_110 }
		set_variable = { BH3_ER_dlv_window = 9 }
	}
	else = {
		set_variable = { BH3_ER_dlv_window = 7 }
	}
}

# 核心II：累计释放 40 次后，把加成变为常驻
BH3_ER_dlv_war = {
	if = {
		limit = {
			has_country_flag = BH3_ER_own_dlv_111
			check_variable = { BH3_ER_dlv_ult_count > 39 }
		}
		set_country_flag = BH3_ER_dlv_permanent
	}
}
