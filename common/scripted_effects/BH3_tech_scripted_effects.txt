BH3_refresh_tech_progressbar = {
	meta_effect = {
		text = {
			set_temp_variable = {
				tech_t_day = [days]
			}
		}
		days = "[GetBH3TechTime]"
	}
	set_temp_variable = {
		tech_r_day = BH3_tech_days_remain
	}
	#log = "tech_r_day 0 [?tech_r_day]"
	if = {
		limit = {
			NOT = {
				check_variable = {
					BH3_tech_days_remain = 0
				}
			}
		}
		set_temp_variable = {
			pbl = 1000
		}
		divide_temp_variable = {
			var = tech_r_day
			value = tech_t_day
		}
		#log = "dvd result [?tech_r_day]"
		multiply_temp_variable = {
			tech_r_day = pbl
		}
		subtract_from_temp_variable = {
			tech_r_day = pbl
		}
		#log = "fal result [?tech_r_day]"
		set_variable = {
			BH3_research_progress_p = -1000
		}
		subtract_from_variable = {
			BH3_research_progress_p = tech_r_day
		}
	}
	else = {
		set_variable = {
			BH3_research_progress_p = 0
		}
	} 
}

### 添加崩坏能研究设施 ###
BH3_Build_HonkaiFacility = {
	every_country = {
		if = {
			limit = { 
				OR = {
					has_facility_specialization = specialization_land
					has_facility_specialization = specialization_air
					has_facility_specialization = specialization_naval
					has_facility_specialization = specialization_nuclear
				}
			}
			set_global_flag = BH3_Honkai_Research_facility_on
		}
	}
	if = {
		limit = { has_global_flag = BH3_Honkai_Research_facility_on }
		ROOT = {
			random_owned_controlled_state = {
				limit = { can_construct_building = BH3_Honkai_Research_facility }
				construct_building_in_random_province = {
					BH3_Honkai_Research_facility = 1
				}
			}
			random_owned_controlled_state = {
				limit = { can_construct_building = BH3_Honkai_Research_facility }
				construct_building_in_random_province = {
					BH3_Honkai_Research_facility = 1
				}
			}
		}
	}
}

#经验满了就替换特质
BH3_remove_valkyrie_level = {
	if = {
		limit = { has_trait = BH3_traits_valkyrie_b }
		remove_unit_leader_trait = BH3_traits_valkyrie_b
		remove_unit_leader_trait = BH3_traits_valkyrie_b2a
		add_unit_leader_trait = BH3_traits_valkyrie_a
	}
	else_if = {
		limit = { has_trait = BH3_traits_valkyrie_a }
		remove_unit_leader_trait = BH3_traits_valkyrie_a
		remove_unit_leader_trait = BH3_traits_valkyrie_a2s
		add_unit_leader_trait = BH3_traits_valkyrie_s
	}
	else_if = {
		limit = { has_trait = BH3_traits_valkyrie_s }
		remove_unit_leader_trait = BH3_traits_valkyrie_s
		remove_unit_leader_trait = BH3_traits_valkyrie_s2ss
		add_unit_leader_trait = BH3_traits_valkyrie_ss
	}
	else_if = {
		limit = { has_trait = BH3_traits_valkyrie_ss }
		remove_unit_leader_trait = BH3_traits_valkyrie_ss
		remove_unit_leader_trait = BH3_traits_valkyrie_ss2sss
		add_unit_leader_trait = BH3_traits_valkyrie_sss
	}
}
