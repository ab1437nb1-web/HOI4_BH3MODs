# 往事乐土·繁星：给敌人染红/黄/蓝 → 我方对敌攻击（红）+ 敌方空海陆减益（黄/蓝）（2026-11-26 第二版）
# 颜色映射（全部染在敌人身上）：
#   红 = 敌方陆军防御下降；我方获得“对特定国家的攻击”加成（自身 buff）
#   黄 = 敌方空军防御/机动下降
#   蓝 = 敌方海军防御下降、海军可见度上升
# 变量：
#   BH3_ER_st_red / _yellow / _blue  敌方颜料剩余天数
#   BH3_ER_st_stack 1104 叠层（0~4）/ BH3_ER_st_dur 颜料持续天数
#   BH3_ER_st_redatk 输出给 DM（army_attack_against_major/minor）

BH3_ER_st_recalc = {
	set_variable = { BH3_ER_st_redatk = 0 }

	# 红色颜料存在 → 对敌攻击
	if = { limit = { check_variable = { BH3_ER_st_red > 0 } } add_to_variable = { BH3_ER_st_redatk = 0.10 } }
	# 1101 红色，热热的：基础强度主要来源
	if = { limit = { has_country_flag = BH3_ER_own_st_1101 } add_to_variable = { BH3_ER_st_redatk = 0.10 } }
	# 1104 黑色：每次染色叠一层
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1104 check_variable = { BH3_ER_st_stack > 0 } }
		set_temp_variable = { st_t = BH3_ER_st_stack }
		multiply_variable = { st_t = 0.03 }
		clamp_variable = { var = st_t min = 0 max = 0.12 }
		add_to_variable = { BH3_ER_st_redatk = st_t }
	}
	# 1107 核心I：敌人身上颜色数 1/2/3 → 对敌攻击 +5%/10%/15%
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1107 }
		set_temp_variable = { st_colors = 0 }
		if = { limit = { check_variable = { BH3_ER_st_red > 0 } } add_to_variable = { st_colors = 1 } }
		if = { limit = { check_variable = { BH3_ER_st_yellow > 0 } } add_to_variable = { st_colors = 1 } }
		if = { limit = { check_variable = { BH3_ER_st_blue > 0 } } add_to_variable = { st_colors = 1 } }
		set_temp_variable = { st_t = st_colors }
		multiply_variable = { st_t = 0.05 }
		add_to_variable = { BH3_ER_st_redatk = st_t }
	}
	# 1109 火焰：红+黄 → 对敌攻击额外 +5%
	if = {
		limit = {
			has_country_flag = BH3_ER_own_st_1109
			check_variable = { BH3_ER_st_red > 0 }
			check_variable = { BH3_ER_st_yellow > 0 }
		}
		add_to_variable = { BH3_ER_st_redatk = 0.05 }
	}

	clamp_variable = { var = BH3_ER_st_redatk min = 0 max = 1.5 }
}

# ---------- 染色（染色 = 给全体敌国挂限时 debuff） ----------
BH3_ER_st_paint_red = {
	set_variable = { BH3_ER_st_red = BH3_ER_st_dur }
	every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_enemy_red days = BH3_ER_st_dur } }
	if = { limit = { has_country_flag = BH3_ER_own_st_1104 check_variable = { BH3_ER_st_stack < 4 } } add_to_variable = { BH3_ER_st_stack = 1 } }
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1105 check_variable = { BH3_ER_st_pp_cd = 0 } }
		add_political_power = 12
		set_variable = { BH3_ER_st_pp_cd = 5 }
	}
	# 红+蓝
	if = {
		limit = { check_variable = { BH3_ER_st_blue > 0 } }
		if = { limit = { has_country_flag = BH3_ER_own_st_1108 } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_rb days = BH3_ER_st_dur } } }
		if = { limit = { has_country_flag = BH3_ER_own_st_1112 } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_total days = BH3_ER_st_dur } } }
		if = {
			limit = { has_country_flag = BH3_ER_own_st_1111 }
			every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_mix days = BH3_ER_st_dur } }
			add_timed_idea = { idea = BH3_ER_st_window days = 5 }
		}
	}
	# 红+黄
	if = {
		limit = { check_variable = { BH3_ER_st_yellow > 0 } }
		if = { limit = { has_country_flag = BH3_ER_own_st_1109 } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_ry days = BH3_ER_st_dur } } }
		if = { limit = { has_country_flag = BH3_ER_own_st_1114 } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_total days = BH3_ER_st_dur } } }
		if = {
			limit = { has_country_flag = BH3_ER_own_st_1111 }
			every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_mix days = BH3_ER_st_dur } }
			add_timed_idea = { idea = BH3_ER_st_window days = 5 }
		}
	}
	BH3_ER_st_recalc = yes
}

# 每日
BH3_ER_st_daily = {
	set_variable = { BH3_ER_st_dur = 8 }
	if = { limit = { has_country_flag = BH3_ER_own_st_1106 } add_to_variable = { BH3_ER_st_dur = 5 } }

	if = { limit = { check_variable = { BH3_ER_st_red > 0 } } subtract_from_variable = { BH3_ER_st_red = 1 } }
	if = { limit = { check_variable = { BH3_ER_st_yellow > 0 } } subtract_from_variable = { BH3_ER_st_yellow = 1 } }
	if = { limit = { check_variable = { BH3_ER_st_blue > 0 } } subtract_from_variable = { BH3_ER_st_blue = 1 } }
	if = { limit = { check_variable = { BH3_ER_st_pp_cd > 0 } } subtract_from_variable = { BH3_ER_st_pp_cd = 1 } }

	# 染色事件
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1101 }
		add_to_variable = { BH3_ER_st_p1 = 1 }
		if = {
			limit = { check_variable = { BH3_ER_st_p1 > 2 } }
			set_variable = { BH3_ER_st_p1 = 0 }
			BH3_ER_st_paint_red = yes
		}
	}
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1102 }
		add_to_variable = { BH3_ER_st_p2 = 1 }
		if = {
			limit = { check_variable = { BH3_ER_st_p2 > 3 } }
			set_variable = { BH3_ER_st_p2 = 0 }
			BH3_ER_st_paint_blue = yes
		}
	}
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1103 }
		add_to_variable = { BH3_ER_st_p3 = 1 }
		if = {
			limit = { check_variable = { BH3_ER_st_p3 > 4 } }
			set_variable = { BH3_ER_st_p3 = 0 }
			BH3_ER_st_paint_yellow = yes
		}
	}

	# 核心I 1107：按颜色数追加敌方 debuff
	if = {
		limit = { has_country_flag = BH3_ER_own_st_1107 }
		set_temp_variable = { st_colors = 0 }
		if = { limit = { check_variable = { BH3_ER_st_red > 0 } } add_to_variable = { st_colors = 1 } }
		if = { limit = { check_variable = { BH3_ER_st_yellow > 0 } } add_to_variable = { st_colors = 1 } }
		if = { limit = { check_variable = { BH3_ER_st_blue > 0 } } add_to_variable = { st_colors = 1 } }
		if = { limit = { check_variable = { st_colors > 0 } } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_mix days = BH3_ER_st_dur } } }
		if = { limit = { check_variable = { st_colors > 1 } } every_enemy_country = { add_timed_idea = { idea = BH3_ER_st_mark_total days = BH3_ER_st_dur } } }
	}

	BH3_ER_st_recalc = yes
}
