######## 计算全球最强女武神
BH3_World_Strongest_Valkyrie_Solve  = {
	find_highest_in_array = {
		array = global.BH3_World_Strongest_Valkyire_Score_Array
		value = v
		index = i
	}
	set_variable = { global.BH3_World_Strongest_Valkyire_Score_Temp = v }
	set_variable = { global.BH3_World_Strongest_Valkyire_Score_index = i }
	for_each_loop = {
		array = global.BH3_World_Strongest_Valkyire_Array
		index = i
		value = v
		if = {
			limit = { check_variable = { i = global.BH3_World_Strongest_Valkyire_Score_index } }
			set_variable = { global.BH3_World_Strongest_Valkyire_Temp = v }
		}
	}
	for_each_loop = {
		array = global.BH3_World_Strongest_Valkyire_Country_Array
		index = i
		value = v
		if = {
			limit = { check_variable = { i = global.BH3_World_Strongest_Valkyire_Score_index } }
			set_variable = { global.BH3_World_Strongest_Valkyire_Country_Temp = v }
		}
	}
	if = {
		limit = { check_variable = { global.BH3_World_Strongest_Valkyire_Score_Temp > global.BH3_World_Strongest_Valkyire_Score_Temp_Prev } }
		#移除最强女武神特质并重新添加
		var:global.BH3_World_Strongest_Valkyire_Country_Temp_Prev = {
			var:global.BH3_World_Strongest_Valkyire_Temp_Prev = {
				if = { 
					limit = { has_trait = BH3_trait_WroldStrongestValkyrie }
					remove_trait = { trait = BH3_trait_WroldStrongestValkyrie }
				}
			}
		}
		#加特质
		var:global.BH3_World_Strongest_Valkyire_Country_Temp = {
			var:global.BH3_World_Strongest_Valkyire_Temp = {
				add_trait = { trait = BH3_trait_WroldStrongestValkyrie }
			}
		}
		set_variable = { global.BH3_World_Strongest_Valkyire_Country_Temp_Prev = global.BH3_World_Strongest_Valkyire_Country_Temp }
		set_variable = { global.BH3_World_Strongest_Valkyire_Temp_Prev = global.BH3_World_Strongest_Valkyire_Temp }
		set_variable = { global.BH3_World_Strongest_Valkyire_Score_Temp_Prev = global.BH3_World_Strongest_Valkyire_Score_Temp }

		set_variable = { global.BH3_World_Strongest_Valkyire_Country = global.BH3_World_Strongest_Valkyire_Country_Temp }
		set_variable = { global.BH3_World_Strongest_Valkyire = global.BH3_World_Strongest_Valkyire_Temp }
		set_variable = { global.BH3_World_Strongest_Valkyire_Score = global.BH3_World_Strongest_Valkyire_Score_Temp }
	}

	clear_array = global.BH3_World_Strongest_Valkyire_Array
	clear_array = global.BH3_World_Strongest_Valkyire_Score_Array
	clear_array = global.BH3_World_Strongest_Valkyire_Country_Array
}
