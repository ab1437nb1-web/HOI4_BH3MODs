# 往事乐土·黄金：能量（政治点）体系（2026-11-25 第二版）
# 普通：按当前政治点给动态修正（每 25 点一档，日结近似）
# 核心I：累计回复 100 政治点触发一次 3 天限时加成
# 核心II：按累计消耗政治点换算终末余音增伤

BH3_ER_gold_daily = {
	set_variable = { BH3_ER_gold_bucket = 0 }
	if = { limit = { political_power > 24 } set_variable = { BH3_ER_gold_bucket = 1 } }
	if = { limit = { political_power > 49 } set_variable = { BH3_ER_gold_bucket = 2 } }
	if = { limit = { political_power > 74 } set_variable = { BH3_ER_gold_bucket = 3 } }

	if = { limit = { political_power > 124 } set_variable = { BH3_ER_gold_bucket = 5 } }
	if = { limit = { political_power > 149 } set_variable = { BH3_ER_gold_bucket = 6 } }
	if = { limit = { political_power > 174 } set_variable = { BH3_ER_gold_bucket = 7 } }
	if = { limit = { political_power > 199 } set_variable = { BH3_ER_gold_bucket = 8 } }
	if = { limit = { political_power > 224 } set_variable = { BH3_ER_gold_bucket = 9 } }
	if = { limit = { political_power > 249 } set_variable = { BH3_ER_gold_bucket = 10 } }
	if = { limit = { political_power > 274 } set_variable = { BH3_ER_gold_bucket = 11 } }
	if = { limit = { political_power > 299 } set_variable = { BH3_ER_gold_bucket = 12 } }
	if = { limit = { political_power > 324 } set_variable = { BH3_ER_gold_bucket = 13 } }
	if = { limit = { political_power > 349 } set_variable = { BH3_ER_gold_bucket = 14 } }
	if = { limit = { political_power > 374 } set_variable = { BH3_ER_gold_bucket = 15 } }
	if = { limit = { political_power > 399 } set_variable = { BH3_ER_gold_bucket = 16 } }
	if = { limit = { political_power > 424 } set_variable = { BH3_ER_gold_bucket = 17 } }
	if = { limit = { political_power > 449 } set_variable = { BH3_ER_gold_bucket = 18 } }
	if = { limit = { political_power > 474 } set_variable = { BH3_ER_gold_bucket = 19 } }
	if = { limit = { political_power > 499 } set_variable = { BH3_ER_gold_bucket = 20 } }
	if = { limit = { political_power > 524 } set_variable = { BH3_ER_gold_bucket = 21 } }
	if = { limit = { political_power > 549 } set_variable = { BH3_ER_gold_bucket = 22 } }
	if = { limit = { political_power > 574 } set_variable = { BH3_ER_gold_bucket = 23 } }
	if = { limit = { political_power > 599 } set_variable = { BH3_ER_gold_bucket = 24 } }
	if = { limit = { political_power > 624 } set_variable = { BH3_ER_gold_bucket = 25 } }
	if = { limit = { political_power > 649 } set_variable = { BH3_ER_gold_bucket = 26 } }
	if = { limit = { political_power > 674 } set_variable = { BH3_ER_gold_bucket = 27 } }
	if = { limit = { political_power > 699 } set_variable = { BH3_ER_gold_bucket = 28 } }
	if = { limit = { political_power > 724 } set_variable = { BH3_ER_gold_bucket = 29 } }
	if = { limit = { political_power > 749 } set_variable = { BH3_ER_gold_bucket = 30 } }

	# 累计回复 / 累计消耗（用档位差近似）
	if = {
		limit = { check_variable = { BH3_ER_gold_bucket > BH3_ER_gold_prev } }
		set_temp_variable = { gd = BH3_ER_gold_bucket }
		subtract_from_variable = { gd = BH3_ER_gold_prev }
		multiply_variable = { gd = 25 }
		add_to_variable = { BH3_ER_gold_gain = gd }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_gold_bucket < BH3_ER_gold_prev } }
		set_temp_variable = { gd = BH3_ER_gold_prev }
		subtract_from_variable = { gd = BH3_ER_gold_bucket }
		multiply_variable = { gd = 25 }
		add_to_variable = { BH3_ER_gold_spend = gd }
	}
	set_variable = { BH3_ER_gold_prev = BH3_ER_gold_bucket }

	# 咏叹限时加成倒计时
	if = {
		limit = { check_variable = { BH3_ER_gold_aria_timer > 0 } }
		subtract_from_variable = { BH3_ER_gold_aria_timer = 1 }
		if = {
			limit = { check_variable = { BH3_ER_gold_aria_timer = 0 } }
			set_variable = { BH3_ER_gold_aria_on = 0 }
		}
	}

	# 累计回复达到 100 → 雷鸣 + 沉醉（限时 3 天）
	if = {
		limit = {
			has_country_flag = BH3_ER_own_gold_207
			check_variable = { BH3_ER_gold_gain > 99 }
		}
		subtract_from_variable = { BH3_ER_gold_gain = 100 }
		set_variable = { BH3_ER_gold_aria_timer = 3 }
		set_variable = { BH3_ER_gold_aria_on = 1 }
	}

	BH3_ER_gold_recalc = yes
}

# 重算输出变量（DM 直接读这些变量）
BH3_ER_gold_recalc = {
	set_variable = { BH3_ER_gold_pp_approx = BH3_ER_gold_bucket }
	multiply_variable = { BH3_ER_gold_pp_approx = 25 }

	# ---- 全伤害 ----
	set_variable = { BH3_ER_gold_atk = 0 }
	if = {
		limit = { has_country_flag = BH3_ER_own_gold_201 }
		set_temp_variable = { gt = BH3_ER_gold_pp_approx }
		multiply_variable = { gt = 0.003 }
		clamp_variable = { var = gt min = 0 max = 1.5 }
		add_to_variable = { BH3_ER_gold_atk = gt }
	}
	if = {
		limit = { has_country_flag = BH3_ER_own_gold_202 }
		set_temp_variable = { gt = BH3_ER_gold_pp_approx }
		multiply_variable = { gt = 0.0025 }
		clamp_variable = { var = gt min = 0 max = 1.25 }
		add_to_variable = { BH3_ER_gold_atk = gt }
	}
	if = {
		limit = { has_country_flag = BH3_ER_own_gold_213 }
		add_to_variable = { BH3_ER_gold_atk = 0.15 }
	}
	if = {
		limit = { has_country_flag = BH3_ER_own_gold_211 }
		set_temp_variable = { gt = BH3_ER_gold_spend }
		multiply_variable = { gt = 0.002 }
		clamp_variable = { var = gt min = 0 max = 1.6 }
		add_to_variable = { BH3_ER_gold_atk = gt }
	}

	# ---- 防御（203） ----
	set_variable = { BH3_ER_gold_def = 0 }
	if = {
		limit = { has_country_flag = BH3_ER_own_gold_203 }
		set_temp_variable = { gt = BH3_ER_gold_pp_approx }
		multiply_variable = { gt = 0.0025 }
		clamp_variable = { var = gt min = 0 max = 0.8 }
		add_to_variable = { BH3_ER_gold_def = gt }
	}

	# ---- 政治点获取 ----
	set_variable = { BH3_ER_gold_pp = 0 }
	if = { limit = { has_country_flag = BH3_ER_own_gold_204 } add_to_variable = { BH3_ER_gold_pp = 0.02 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_205 } add_to_variable = { BH3_ER_gold_pp = 0.05 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_206 } add_to_variable = { BH3_ER_gold_pp = 0.03 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_209 } add_to_variable = { BH3_ER_gold_pp = 0.05 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_212 } add_to_variable = { BH3_ER_gold_pp = 0.10 } }

	# ---- 对主要/次要国家攻击 ----
	set_variable = { BH3_ER_gold_maj = 0 }
	if = { limit = { has_country_flag = BH3_ER_own_gold_208 } add_to_variable = { BH3_ER_gold_maj = 0.15 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_210 } add_to_variable = { BH3_ER_gold_maj = 0.20 } }
	if = { limit = { has_country_flag = BH3_ER_own_gold_214 } add_to_variable = { BH3_ER_gold_maj = 0.20 } }
	if = { limit = { check_variable = { BH3_ER_gold_aria_on = 1 } } add_to_variable = { BH3_ER_gold_maj = 0.20 } }
	set_variable = { BH3_ER_gold_min = BH3_ER_gold_maj }
}
