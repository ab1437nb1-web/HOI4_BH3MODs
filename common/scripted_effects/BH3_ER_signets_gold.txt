# 往事乐土·黄金：能量（政治点）体系
# 变量：BH3_ER_gold_atk / _def / _pp / _maj / _min（常驻累加）
# 窗口：BH3_ER_gold_window（天数）+ BH3_ER_gold_w_*（窗口期间临时加值，到期扣回）

# 核心I：咏叹（消耗 50 政治点，7 天窗口）
BH3_ER_gold_aria = {
	if = {
		limit = { check_variable = { BH3_ER_gold_w_on = 0 } }
		add_to_variable = { BH3_ER_gold_atk = 0.30 }
		add_to_variable = { BH3_ER_gold_maj = 0.20 }
		add_to_variable = { BH3_ER_gold_min = 0.20 }
		set_variable = { BH3_ER_gold_w_atk = 0.30 }
		set_variable = { BH3_ER_gold_w_maj = 0.20 }
		set_variable = { BH3_ER_gold_w_min = 0.20 }
		set_variable = { BH3_ER_gold_w_on = 1 }
	}
	set_variable = { BH3_ER_gold_window = 7 }
}

# 核心II：余音（消耗 100 政治点，14 天窗口）

		set_variable = { BH3_ER_gold_w_maj = 0.30 }
		set_variable = { BH3_ER_gold_w_min = 0.30 }
		set_variable = { BH3_ER_gold_w_on = 1 }
	}

}

# 窗口到期：把临时加值扣回
BH3_ER_gold_window_end = {
	if = {
		limit = { check_variable = { BH3_ER_gold_w_on = 1 } }
		subtract_from_variable = { BH3_ER_gold_atk = BH3_ER_gold_w_atk }
		subtract_from_variable = { BH3_ER_gold_maj = BH3_ER_gold_w_maj }
		subtract_from_variable = { BH3_ER_gold_min = BH3_ER_gold_w_min }
		set_variable = { BH3_ER_gold_w_on = 0 }
		set_variable = { BH3_ER_gold_w_atk = 0 }
		set_variable = { BH3_ER_gold_w_maj = 0 }
		set_variable = { BH3_ER_gold_w_min = 0 }
	}
}
