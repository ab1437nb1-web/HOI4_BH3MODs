# 商店系统底层门控：任何消耗刻印池的逻辑（商店/事件等）在使用前必须检查这里
# BH3_ER_count_pool 会重算 BH3_ER_pool_left（可抽取的未拥有非核心刻印数）

BH3_ER_pool_has_signet = {
	check_variable = { BH3_ER_pool_left > 0 }
}

# 商店出牌前的最低保留量：至少留 4 枚给后续抽取层
BH3_ER_pool_safe_for_shop = {
	check_variable = { BH3_ER_pool_left > 3 }
}

# 抽取层保底：只要池子非空就必须能抽出东西
BH3_ER_draw_layer_guaranteed = {
	OR = {
		check_variable = { BH3_ER_pool_left > 0 }
		check_variable = { BH3_ER_draw_fallback = 1 }
	}
}

# 商店刷新是否可用：空梦之币 >= 下次刷新费用（100 起，每次翻倍）
BH3_ER_can_shop_refresh = {
	if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 0 } }
		check_variable = { BH3_ER_rv_coin > 99 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 1 } }
		check_variable = { BH3_ER_rv_coin > 199 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 2 } }
		check_variable = { BH3_ER_rv_coin > 399 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 3 } }
		check_variable = { BH3_ER_rv_coin > 799 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 4 } }
		check_variable = { BH3_ER_rv_coin > 1599 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 5 } }
		check_variable = { BH3_ER_rv_coin > 3199 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 6 } }
		check_variable = { BH3_ER_rv_coin > 6399 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 7 } }
		check_variable = { BH3_ER_rv_coin > 12799 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 8 } }
		check_variable = { BH3_ER_rv_coin > 25599 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 9 } }
		check_variable = { BH3_ER_rv_coin > 51199 }
	}
	else_if = {
		limit = { check_variable = { BH3_ER_shop_refresh_count = 10 } }
		check_variable = { BH3_ER_rv_coin > 102399 }
	}
	else = {
		check_variable = { BH3_ER_rv_coin > 204799 }
	}
}
