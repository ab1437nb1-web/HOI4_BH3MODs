on_actions = {

	# 旧存档/异常情况下补一次初始化
	on_daily = {
		effect = {
			every_country = {
				limit = {
					has_variable = BH3_ER_dlv_window
					check_variable = { BH3_ER_dlv_window > 0 }
				}
				subtract_from_variable = { BH3_ER_dlv_window = 1 }
			}
			every_country = {
				limit = { has_country_flag = BH3_ElysianRealm_unlocked }
				BH3_ER_gold_daily = yes
			}
			every_country = {
				limit = {
					has_country_flag = BH3_ElysianRealm_unlocked
					NOT = { has_variable = BH3_ER_floor }
				}
				BH3_ER_init_run = yes
			}
		}
	}
}
