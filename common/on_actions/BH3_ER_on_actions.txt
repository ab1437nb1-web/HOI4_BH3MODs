on_actions = {

	# 旧存档/异常情况下补一次初始化
	on_daily = {
		effect = {
			every_country = {
				limit = {
					has_variable = BH3_ER_dlv_window
					check_variable = { BH3_ER_dlv_window > 0 }
				}
				subtract_from_variable = { BH3_ER_dlv_window = 1 }
			}
			every_country = {
				limit = { has_country_flag = BH3_ElysianRealm_unlocked }
				BH3_ER_gold_daily = yes
				BH3_ER_setsuna_daily = yes
				BH3_ER_surr_daily = yes
				BH3_ER_fs_daily = yes
				if = { limit = { check_variable = { BH3_ER_surr_delta > 0 } } BH3_ER_az_on_loss = yes }
				BH3_ER_az_daily = yes
				BH3_ER_bd_daily = yes
				BH3_ER_ds_daily = yes
				BH3_ER_hx_daily = yes
				BH3_ER_if_daily = yes
				BH3_ER_db_daily = yes
				BH3_ER_st_daily = yes
			}
			every_country = {
				limit = {
					has_country_flag = BH3_ElysianRealm_unlocked
					NOT = { has_variable = BH3_ER_floor }
				}
				BH3_ER_init_run = yes
			}
		}
	}
}

# 核袭击拦截：目标国打开拦截窗口时，被炸方（FROM 的 owner）消耗 1 次无敌储存
on_nuke_drop = {
	effect = {
		if = {
			limit = {
				FROM = {
					owner = { has_country_flag = BH3_ER_fs_invuln_active }
				}
			}
			FROM = {
				owner = {
					if = {
						limit = { check_variable = { BH3_ER_fs_charges > 0 } }
						subtract_from_variable = { BH3_ER_fs_charges = 1 }
					}
				}
			}
		}
	}
}
