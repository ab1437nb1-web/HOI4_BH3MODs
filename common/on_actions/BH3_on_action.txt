	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM = {
						NOT = { is_in_faction_with = BHS }
					}
					ROOT = {
						is_in_faction_with = BHS
					}
					FROM.FROM = {
						NOT = { has_dynamic_modifier = { modifier = BHSinvasion_effect } }
					}
				}
				FROM.FROM = {
					add_core_of = ROOT
				}
				# 吃工业
				every_country = {
					limit = { is_in_faction_with = BHS }
					add_offsite_building = { type = arms_factory level = PREV.FROM.FROM.arms_factory_level }
					add_offsite_building = { type = industrial_complex level = PREV.FROM.FROM.industrial_complex_level }
					#造船厂吃不到
					add_offsite_building = { type = dockyard level = PREV.FROM.FROM.building_level@dockyard }
				}
				# 拆工业
				FROM.FROM = {
					remove_building = {
						type = arms_factory
						level = arms_factory_level
					}
					remove_building = {
						type = industrial_complex
						level = industrial_complex_level
					}
					remove_building = {
						type = dockyard
						level = building_level@dockyard
					}
				}
				
				FROM.FROM = {
					#添加省份debuff
					add_dynamic_modifier = { modifier = BHSinvasion_effect }
					# 统计崩坏侵蚀人口
					set_temp_variable = { HonkaiImpact_Cause_Population_Casualty_m = state_population_k }
					divide_temp_variable = { HonkaiImpact_Cause_Population_Casualty_m = 1000 } 
					add_to_variable = { global.HonkaiImpact_Cause_Population_Casualty = HonkaiImpact_Cause_Population_Casualty_m }
					# 计算当地人口转为崩坏死士,如果没有超过限制数量
					BHS_SpawnSishiBHS_On_Control_State = yes
				}
				#统计剩余人口（百万人）
				subtract_from_variable = { global.HonkaiImpact_Remain_Population_Casualty = HonkaiImpact_Cause_Population_Casualty_m }
				# 计算侵蚀人口百分比
				set_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = global.HonkaiImpact_Cause_Population_Casualty }
				divide_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = global.HonkaiImpact_Population_Total }
				multiply_variable = { global.HonkaiImpact_Cause_Population_Casualty_p = 100 }
				if = {
					limit = { NOT = { has_global_flag = HonkaiImpact_Cause_Population_Casualty_p_at_least_70 } }
					if = { 
						limit = { check_variable = { global.HonkaiImpact_Cause_Population_Casualty_p > 70 } }
						set_global_flag = HonkaiImpact_Cause_Population_Casualty_p_at_least_70
					}
				}
			}
		}
	}
