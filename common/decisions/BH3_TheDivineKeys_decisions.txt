	BH3_ThousandKey_decision = {
		priority = 99
		icon = GFX_decision_ThousandKey
		visible = {
			NOT = {

			}
		}
		available = {

			OR = {
				has_equipment = { ThousandKey_equipment > 49 }

			}
		}
		ai_will_do = {
			base = 0
		}
		fire_only_once = no
		cost = 0
		modifier = {
			supply_consumption_factor = 0.5
			supply_node_range = 10
			army_speed_factor = 0.5
			naval_speed_factor = 0.5
			production_speed_buildings_factor = 1
			productivity_growth_modifier = 100
			expected_mil_modifier = 1
		}
		days_remove = 60
		complete_effect = {
			custom_effect_tooltip = BH3_ThousandKey_decision_tt
			# 只刮生产率
			if = {
				limit = { has_dynamic_modifier = no_fuel_penalties_modifier }
				hidden_effect = {
					set_variable = { ROOT.BH3_productivity_state_var_SUM = 0 }
					every_controlled_state = {
						add_to_variable = {  ROOT.BH3_productivity_state_var_SUM = productivity_state_var }
						set_variable = { productivity_state_var = 0 }
					}
					capital_scope = {
						set_variable = { productivity_state_var =  ROOT.BH3_productivity_state_var_SUM }
					}
				}
			}
			# 没有该modify，就刮人口
			else = {
				hidden_effect = {
					set_variable = { ROOT.BH3_population_state_var_SUM = 0 }
					every_controlled_state = {
						add_to_variable = {  ROOT.BH3_population_state_var_SUM = THIS.state_population }
						add_manpower = -999999999
					}
					capital_scope = {
						add_manpower = ROOT.BH3_population_state_var_SUM
					}
				}
			}

			
			every_controlled_state = {
				limit = {  THIS = {  arms_factory > 0 } }
				ROOT = {
					add_offsite_building = {
						type = arms_factory
						level = PREV.arms_factory_level
					}
				}
                THIS = {
                    set_building_level = {
                        type = arms_factory
                        level = 0
                    }
                }
			}
			every_controlled_state = {
				limit = {  THIS = {  industrial_complex > 0 } }
				ROOT = {
					add_offsite_building = {
						type = industrial_complex
						level = PREV.industrial_complex_level
					}
				}
                THIS = {
                    set_building_level = {
                        type = industrial_complex
                        level = 0
                    }
                }
			}
			every_controlled_state = {
				limit = {  THIS = {  dockyard > 0 } }
				ROOT = {
					add_offsite_building = {
						type = dockyard
						level = PREV.building_level@dockyard
					}
				}
                THIS = {
                    set_building_level = {
                        type = dockyard
                        level = 0
                    }
                }
			}
			add_equipment_to_stockpile = {
				type = ThousandKey_equipment
				amount = -50
			}
		}
	}
